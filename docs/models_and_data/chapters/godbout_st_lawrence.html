mod_ecoregion <- readRDS(here::here("models/mod_ecoregion.rds"))


# for plotting to do the back transform
exp_trans <- function(dat){

  dat |>
    mutate(
      Predicted = exp(Predicted)-0.01,
           CI_low = exp(CI_low)-0.01,
           CI_high = exp(CI_high)-0.01,
           CI_high = ifelse(CI_high>1.1, 1.1, CI_high)
    )|>
               mutate(trajectory = 
                        gsub(study[1], "", trajectory))

}


re_ecoregion <- ranef(mod_ecoregion)$cond$study |>
  as_tibble(rownames = "study")  |> 
  filter(study == params$dataset_name) |> 
  pull(`(Intercept)`)


dat_fixed_ecoregion <- estimate_relation(mod_ecoregion,
                               include_random = FALSE,
                  length = 200,
                  data = kelptime_study) |>
  mutate(Predicted = Predicted + re_ecoregion,
         CI_high = CI_high + re_ecoregion,
         CI_low = CI_low + re_ecoregion) |>
  exp_trans()

dat_traj_ecoregion <- estimate_relation(mod_ecoregion,
                              include_random = TRUE,
                  length = 200,
                  data = kelptime_study) |>
    exp_trans()


ggplot(dat_traj_ecoregion, aes(x = year)) +
  geom_line(aes(y = Predicted,
                color = trajectory), alpha = 0.5) +
  geom_point(aes(y = focal_std_by_ecoregion,
                 group = trajectory,
                 color = trajectory), alpha = 0.5,
             data = kelptime_study |>
               mutate(trajectory = 
                        gsub(study[1], "", trajectory))) +
